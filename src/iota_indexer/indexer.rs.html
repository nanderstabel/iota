35impl Indexer {
36    pub async fn start_writer(
37        config: &IngestionConfig,
38        store: PgIndexerStore,
39        metrics: IndexerMetrics,
40    ) -> Result<(), IndexerError> {
41        let snapshot_config = SnapshotLagConfig::default();
42        Indexer::start_writer_with_config(
43            config,
44            store,
45            metrics,
46            snapshot_config,
47            PruningOptions::default(),
48            CancellationToken::new(),
49        )
50        .await
51    }
52
53    pub async fn start_writer_with_config(
54        config: &IngestionConfig,
55        store: PgIndexerStore,
56        metrics: IndexerMetrics,
57        snapshot_config: SnapshotLagConfig,
58        pruning_options: PruningOptions,
59        cancel: CancellationToken,
60    ) -> Result<(), IndexerError> {
61        info!(
62            "IOTA Indexer Writer (version {:?}) started...",
63            env!("CARGO_PKG_VERSION")
64        );
65
66        info!("IOTA Indexer Writer config: {config:?}",);
67
68        let primary_watermark = store
69            .get_latest_checkpoint_sequence_number()
70            .await
71            .expect("Failed to get latest tx checkpoint sequence number from DB")
72            .map(|seq| seq + 1)
73            .unwrap_or_default();
74        let extra_reader_options = ReaderOptions {
75            batch_size: config.checkpoint_download_queue_size,
76            timeout_secs: config.checkpoint_download_timeout,
77            data_limit: config.checkpoint_download_queue_size_bytes,
78            ..Default::default()
79        };
80
81        // Start objects snapshot processor, which is a separate pipeline with its
82        // ingestion pipeline.
83        let (object_snapshot_worker, object_snapshot_watermark) = start_objects_snapshot_handler(
84            store.clone(),
85            metrics.clone(),
86            snapshot_config,
87            cancel.clone(),
88        )
89        .await?;
90
91        if let Some(epochs_to_keep) = pruning_options.epochs_to_keep {
92            info!(
93                "Starting indexer pruner with epochs to keep: {}",
94                epochs_to_keep
95            );
96            assert!(epochs_to_keep > 0, "Epochs to keep must be positive");
97            let pruner: Pruner = Pruner::new(store.clone(), epochs_to_keep, metrics.clone())?;
98            spawn_monitored_task!(pruner.start(CancellationToken::new()));
99        }
100
101        // If we already have chain identifier indexed (i.e. the first checkpoint has
102        // been indexed), then we persist protocol configs for protocol versions
103        // not yet in the db. Otherwise, we would do the persisting in
104        // `commit_checkpoint` while the first cp is being indexed.
105        if let Some(chain_id) = IndexerStore::get_chain_identifier(&store).await? {
106            store.persist_protocol_configs_and_feature_flags(chain_id)?;
107        }
108
109        let mut executor = IndexerExecutor::new(
110            ShimIndexerProgressStore::new(vec![
111                ("primary".to_string(), primary_watermark),
112                ("object_snapshot".to_string(), object_snapshot_watermark),
113            ]),
114            1,
115            DataIngestionMetrics::new(&Registry::new()),
116            cancel.child_token(),
117        );
118        let worker = new_handlers(store, metrics, primary_watermark, cancel.clone()).await?;
119        let worker_pool = WorkerPool::new(
120            worker,
121            "primary".to_string(),
122            config.checkpoint_download_queue_size,
123            Default::default(),
124        );
125
126        executor.register(worker_pool).await?;
127
128        let worker_pool = WorkerPool::new(
129            object_snapshot_worker,
130            "object_snapshot".to_string(),
131            config.checkpoint_download_queue_size,
132            Default::default(),
133        );
134        executor.register(worker_pool).await?;
135        info!("Starting data ingestion executor...");
136        executor
137            .run(
138                config
139                    .sources
140                    .data_ingestion_path
141                    .clone()
142                    .unwrap_or(tempfile::tempdir().unwrap().into_path()),
143                config
144                    .sources
145                    .remote_store_url
146                    .as_ref()
147                    .map(|url| url.as_str().to_owned()),
148                vec![],
149                extra_reader_options,
150            )
151            .await?;
152        Ok(())
153    }
154
155    pub async fn start_reader(
156        config: &JsonRpcConfig,
157        registry: &Registry,
158        connection_pool: ConnectionPool,
159    ) -> Result<(), IndexerError> {
160        info!(
161            "IOTA Indexer Reader (version {:?}) started...",
162            env!("CARGO_PKG_VERSION")
163        );
164        let indexer_reader = IndexerReader::new(connection_pool);
165        let handle = build_json_rpc_server(registry, indexer_reader, config)
166            .await
167            .expect("Json rpc server should not run into errors upon start.");
168        tokio::spawn(async move { handle.stopped().await })
169            .await
170            .expect("Rpc server task failed");
171
172        Ok(())
173    }
174    pub async fn start_analytical_worker<
175        S: IndexerAnalyticalStore + Clone + Send + Sync + 'static,
176    >(
177        store: S,
178        metrics: IndexerMetrics,
179    ) -> Result<(), IndexerError> {
180        info!(
181            "IOTA Indexer Analytical Worker (version {:?}) started...",
182            env!("CARGO_PKG_VERSION")
183        );
184        let mut processor_orchestrator = ProcessorOrchestrator::new(store, metrics);
185        processor_orchestrator.run_forever().await;
186        Ok(())
187    }
188}

190struct ShimIndexerProgressStore {
191    watermarks: HashMap<String, CheckpointSequenceNumber>,
192}
193
194impl ShimIndexerProgressStore {
195    fn new(watermarks: Vec<(String, CheckpointSequenceNumber)>) -> Self {
196        Self {
197            watermarks: watermarks.into_iter().collect(),
198        }
199    }
200}
201
202#[async_trait]
203impl ProgressStore for ShimIndexerProgressStore {
204    type Error = IndexerError;
205
206    async fn load(&mut self, task_name: String) -> Result<CheckpointSequenceNumber, Self::Error> {
207        Ok(*self.watermarks.get(&task_name).expect("missing watermark"))
208    }
209
210    async fn save(&mut self, _: String, _: CheckpointSequenceNumber) -> Result<(), Self::Error> {
211        Ok(())
212    }
213}
